% FAANGPath Simple Template: EMPTY PLACEHOLDER SKELETON (hand-edited LaTeX alternative)
% Based on the public FAANGPath simple template, hardened with the flags of ../ats-preamble.tex. render_resume.py does
% not use this file; it is the manual route when a resume must be tuned by hand.
% Fill the ALL-CAPS placeholders. Build: /Library/TeX/texbin/pdflatex resume_faangpath.tex
%
% Changes from the upstream template:
%  - Added the proven hardening: \pdfinterwordspaceon at document start (real space glyphs,
%    so PDFKit keeps interword spaces), ragged2e \RaggedRight, no hyphenation
%    (\hyphenpenalty and \exhyphenpenalty = 10000).
%  - Skills are a plain list, not a tabular: no tables anywhere.
%  - Dates sit inline after a bar ("TITLE | MON YYYY - MON YYYY"), not flush right with
%    \hfill: flush-right dates are read as a separate column by pdftotext and PDFKit.
%  - Date ranges use a spaced hyphen (`Aug 2023 - May 2026`), never -- or ---.
%
% ATS notes (still true):
%  - glyphtounicode + T1 encoding: every glyph extracts as real text.
%  - Single column, standard sections, no images, no text boxes.
%  - All \href links carry https://.
%  - Class quirk: with THREE \address lines the class prints the third line FIRST;
%    with TWO lines (as below) the order is normal. Stick to two.
%------------------------------------------------------------------------------------
\documentclass{resume} % Uses resume.cls in this folder

\usepackage[left=0.4in,top=0.4in,right=0.4in,bottom=0.4in]{geometry} % Page margins
\usepackage[T1]{fontenc}          % ATS: clean character encoding in the PDF
\input{glyphtounicode}\pdfgentounicode=1 % ATS: guarantee glyph -> Unicode mapping
\usepackage{ragged2e}             % \RaggedRight: greedy ragged-right line breaking
\hyphenpenalty=10000              % no hyphenation: hyphen breaks split ATS keywords
\exhyphenpenalty=10000
\setlength{\RaggedRightRightskip}{0pt plus 3em}
\tolerance=1000
\emergencystretch=3em
\RaggedRight
% Real space glyphs between words. resume.cls prints the name and address inside
% \begin{document}, so switch it on in the begin-document hook, before the header prints.
\AtBeginDocument{\ifdefined\pdfinterwordspaceon\pdfinterwordspaceon\fi}

\newcommand{\tab}[1]{\hspace{.2667\textwidth}\rlap{#1}} % (template helpers, keep)
\newcommand{\itab}[1]{\hspace{0em}\rlap{#1}}

%--------------------------------- HEADER -------------------------------------------
\name{FIRSTNAME LASTNAME} % Prints centered, uppercase, large
% Two centered contact lines; "\\" inside a line becomes a diamond separator.
\address{+1 (000) 000-0000 \\ CITY, ST}
\address{%
  \href{mailto:EMAIL@EXAMPLE.TEST}{EMAIL@EXAMPLE.TEST} \\
  \href{https://linkedin.com/in/USERNAME}{linkedin.com/in/USERNAME} \\
  \href{https://github.com/USERNAME}{github.com/USERNAME} \\
  \href{https://YOURSITE.COM}{YOURSITE.COM}}
